" + "Latitude: " + currentLatitude.toFixed(6) + "
" + "Longitude: " + currentLongitude.toFixed(6); resolve({ latitude: currentLatitude, longitude: currentLongitude }); }, function(error) { currentLatitude = null; currentLongitude = null; let text = "❌ GPS гирифта нашуд."; if (error.code === 1) { text = "❌ Иҷозаи Location рад шудааст.
